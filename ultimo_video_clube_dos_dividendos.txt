            // localStorage helpers for transcript view counting
            function getViewsToday(key, today) {
                try {
                    const raw = localStorage.getItem(key);
                    if (!raw) return { date: '', count: 0, videoIds: [] };
                    const data = JSON.parse(raw);
                    if (data.date !== today) return { date: '', count: 0, videoIds: [] };
                    return data;
                } catch (e) {
                    return { date: '', count: 0, videoIds: [] };
                }
            }

            function recordView(key, videoId, today) {
                try {
                    const views = getViewsToday(key, today);
                    if (views.date === today && views.videoIds.includes(videoId)) return views.count;
                    const newData = {
                        date: today,
                        count: (views.date === today ? views.count : 0) + 1,
                        videoIds: views.date === today ? [...views.videoIds, videoId] : [videoId]
                    };
                    localStorage.setItem(key, JSON.stringify(newData));
                    return newData.count;
                } catch (e) {
                    return 0;
                }
            }

            function shouldApplyLimit(key, videoId) {
                const today = new Date().toISOString().slice(0, 10);
                const views = getViewsToday(key, today);

                // If user has already viewed transcript for this video today, don't apply limit
                if (views.date === today && views.videoIds.includes(videoId)) return false;

                if (views.count >= GEO_LIMIT.maxFreeViews) {
                    // User has already exceeded the limit today
                    return true;
                }

                const count = recordView(key, videoId, today);
                return false;
            }

            function downloadSubtitles(format) {
                if (!savedTranscription) {
                    return;
                }
                function formatTime(seconds, format) {
                    const date = new Date(seconds * 1000);
                    const hh = String(Math.floor(seconds / 3600)).padStart(2, '0');
                    const mm = String(Math.floor((seconds % 3600) / 60)).padStart(2, '0');
                    const ss = String(Math.floor(seconds % 60)).padStart(2, '0');
                    const ms = String(Math.round((seconds % 1) * 1000)).padStart(3, '0');

                    const separator = format === 'srt' ? ',' : '.';
                    return `${hh}:${mm}:${ss}${separator}${ms}`;
                }

                let content = '';
                const ext = format.toLowerCase();

                if (ext === 'vtt') {
                    content = 'WEBVTT\n\n';

                    savedTranscription.forEach((sub, index) => {
                        const start = formatTime(parseFloat(sub.s), 'vtt');
                        const end = formatTime(parseFloat(sub.e), 'vtt');
                        content += `${start} --> ${end}\n${sub.t}\n\n`;
                    });
                } else if (ext === 'srt') {
                    savedTranscription.forEach((sub, index) => {
                        const start = formatTime(parseFloat(sub.s), 'srt');
                        const end = formatTime(parseFloat(sub.e), 'srt');
                        content += `${index + 1}\n${start} --> ${end}\n${sub.t}\n\n`;
                    });
                } else if (ext === 'txt') {
                    content = extractTranscriptFromHTML(true)
                } else {
                    throw new Error('Unsupported format. Use "srt" or "vtt" or "txt"');
                }

                const blob = new Blob([content], { type: 'text/plain;charset=utf-8' });
                const url = URL.createObjectURL(blob);

                const link = document.createElement('a');
                link.href = url;
                link.download = `Ir8dALr-e-4.${ext}`;
                document.body.appendChild(link);
                link.click();
                document.body.removeChild(link);

                URL.revokeObjectURL(url);
            }

            function getHTMLFromRawTranscript(rawTranscript, videoId) {

                const scriptObjArr = [], timeUpperLimit = 60, charInitLimit = 300, charUpperLimit = 500;
                let loop = 0, chars = [], charCount = 0, timeSum = 0, tempObj = {}, remaining = {};

                // Sum-up to either total 60 seconds or 300 chars.
                Array.from(rawTranscript).forEach((obj, i, arr) => {
                    if (obj.s !== undefined && obj.t !== undefined) {
                        obj.start = obj.s;
                        obj.text = obj.t;
                    }

                    // Check Remaining Text from Prev Loop
                    if (remaining.start && remaining.text) {
                        tempObj.start = remaining.start;
                        chars.push(remaining.text);
                        remaining = {}; // Once used, reset to {}
                    }

                    // Initial Loop: Set Start Time
                    if (loop == 0) {
                        tempObj.start = (remaining.start) ? remaining.start : obj.start;
                    }

                    loop++;

                    const startSeconds = Math.round(tempObj.start);
                    const seconds = Math.round(obj.start);
                    timeSum = (seconds - startSeconds);
                    charCount += obj.text.length;
                    chars.push(obj.text);

                    if (i == arr.length - 1) {
                        tempObj.text = chars.join(" ").replace(/\n/g, " ");
                        scriptObjArr.push(tempObj);
                        resetNums();
                        return;
                    }

                    if (timeSum > timeUpperLimit) {
                        tempObj.text = chars.join(" ").replace(/\n/g, " ");
                        scriptObjArr.push(tempObj);
                        resetNums();
                        return;
                    }

                    if (charCount > charInitLimit) {

                        if (charCount < charUpperLimit) {
                            if (obj.text.includes(".")) {

                                const splitStr = obj.text.split(".");

                                // Case: the last letter is . => Process regulary
                                if (splitStr[splitStr.length - 1].replace(/\s+/g, "") == "") {
                                    tempObj.text = chars.join(" ").replace(/\n/g, " ");
                                    scriptObjArr.push(tempObj);
                                    resetNums();
                                    return;
                                }

                                // Case: . is in the middle
                                // 1. Get the (length - 2) str, then get indexOf + str.length + 1, then substring(0,x)
                                // 2. Create remaining { text: str.substring(x), start: obj.start } => use the next loop
                                const lastText = splitStr[splitStr.length - 2];
                                const substrIndex = obj.text.indexOf(lastText) + lastText.length + 1;
                                const textToUse = obj.text.substring(0, substrIndex);
                                remaining.text = obj.text.substring(substrIndex);
                                remaining.start = obj.start;

                                // Replcae arr element
                                chars.splice(chars.length - 1, 1, textToUse)
                                tempObj.text = chars.join(" ").replace(/\n/g, " ");
                                scriptObjArr.push(tempObj);
                                resetNums();
                                return;

                            } else {
                                // Move onto next loop to find .
                                return;
                            }
                        }

                        tempObj.text = chars.join(" ").replace(/\n/g, " ");
                        scriptObjArr.push(tempObj);
                        resetNums();
                    }
                })

                return Array.from(scriptObjArr).map((obj, idx) => {
                    const t = Math.round(obj.start);
                    const hhmmss = convertIntToHms(t);
                    const timeDisplay = videoId
                        ? `<a href="https://youtube.com/watch?v=${videoId}&t=${t}s" target="_blank">${hhmmss}</a>`
                        : hhmmss;
                    const html = `
                    <div class="transcript-group-box">
                        <div class="transcript-time">${timeDisplay}</div>
                        <div class="transcript-text">${obj.text}</div>
                    </div>`
                    return html
                }).join("");

                function resetNums() {
                    loop = 0, chars = [], charCount = 0, timeSum = 0, tempObj = {};
                }

                function convertIntToHms(num) {
                    const h = (num < 3600) ? 14 : 12;
                    return (new Date(num * 1000).toISOString().substring(h, 19)).toString();
                }
            }

            let currentPercent = 0;

            const percentElement = document.getElementById('percent');

            const radius = 90;

            function updateProgress(percent) {
                percent = Math.min(100, Math.max(0, percent));

                percentElement.textContent = Math.round(percent) + '%';

                const offset = circumference - (percent / 100) * circumference;

            function startAnimation() {
                currentPercent = 0;
                updateProgress(0);
                loadingStatus.textContent = "Carregando";
                stopPulsing();

                if (animationInterval) {
                    clearInterval(animationInterval);
                }

                const targetPercent = 95;
                const duration = 10000;
                const interval = 50;
                const steps = duration / interval;
                const increment = targetPercent / steps;

                animationInterval = setInterval(() => {
                    currentPercent += increment;

                    if (currentPercent >= targetPercent) {
                        currentPercent = targetPercent;
                        updateProgress(currentPercent);
                        clearInterval(animationInterval);
                    } else {
                        updateProgress(currentPercent);
                    }
                }, interval);
            }

            // ========== File Upload Transcript Polling ==========
            async function pollUploadResult(fileVideoId) {
                const apiURL = "https://yt-to-text.com/api";
                const maxAttempts = 150; // ~10 min
                const pollInterval = 4000;
                const largeFileDelay = 15000;
                const mainTranscriptContent = $('#main-transcript-content');

                const largeFileTimeout = setTimeout(() => {
                    loadingStatus.textContent = "Arquivo grande. Isso pode levar um pouco mais de tempo.\nObrigado pela sua paciência.";
                    dotsElement.classList.remove('animating');
                }, largeFileDelay);

                for (let attempt = 0; attempt < maxAttempts; attempt++) {
                    await new Promise(r => setTimeout(r, pollInterval));
                    try {
                        const resp = await fetch(apiURL + '/p/v1/GetUploadResult', {
                            method: 'POST',
                            headers: { 'Content-Type': 'application/json', 'X-App-Version': 'tubetranscript' },
                            body: JSON.stringify({ video_id: fileVideoId }),
                        });

                        if (resp.status === 404) continue;

                        if (!resp.ok) {
                            clearInterval(animationInterval);
                            clearTimeout(largeFileTimeout);
                            mainTranscriptContent.html(`<div class="text-center text-danger">${"Algo deu errado. Tente novamente mais tarde"}</div>`);
                            return;
                        }

                        const data = await resp.json();
                        const tr = data.transcriptions && data.transcriptions[0];
                        if (!tr) continue;

                        if (tr.progressPercent !== undefined) updateProgress(tr.progressPercent);

                        if (tr.status === 'READY') {
                            updateProgress(100);
                            clearInterval(animationInterval);
                            clearTimeout(largeFileTimeout);
                            if (tr.data && tr.data.transcripts && tr.data.transcripts.length > 0) {
                                savedTranscription = tr.data.transcripts;
                                mainTranscriptContent.html(getHTMLFromRawTranscript(tr.data.transcripts, ''));
                            } else {
                                mainTranscriptContent.html(`<div class="text-center text-danger">${"Algo deu errado. Tente novamente mais tarde"}</div>`);
                            }
                            return;
                        }

                        if (tr.status && (tr.status.startsWith('FAILED') || tr.status === 'CANCELED')) {
                            clearInterval(animationInterval);
                            clearTimeout(largeFileTimeout);
                            mainTranscriptContent.html(`<div class="text-center text-danger">${"Algo deu errado. Tente novamente mais tarde"}</div>`);
                            return;
                        }
                    } catch (e) {
                        // network error, keep polling
                    }
                }

                clearInterval(animationInterval);
                clearTimeout(largeFileTimeout);
                mainTranscriptContent.html(`<div class="text-center text-danger">${"A geração do resumo está demorando muito"}</div>`);
            }

            // Main function to fetch and render summary
            async function fetchAndRenderSummary(videoId, languageCode = "auto") {
                const maxAttempts = 60; // 5 minutes max (60 * 5s = 300s)
                const pollInterval = 5000; // 5 seconds
                const apiURL = "https://yt-to-text.com/api";

                // Show loading, hide everything else
                $('#summary-promo').hide();
                $('#summary-error').hide();
                $('#summary-pro-required').hide();
                $('#summary-result').hide();
                $('#summary-loading').show();

                // Reset progress
                updateSummaryProgress(0);

                // Create abort controller for cleanup
                summaryAbortController = new AbortController();

                try {
                    for (let attempt = 0; attempt < maxAttempts; attempt++) {
                        const response = await fetch(apiURL + "/p/v1/GetSummary", {
                            method: "POST",
                            body: JSON.stringify({ video_id: videoId, language_code: languageCode }),
                            headers: {
                                "Content-Type": "application/json",
                                "X-App-Version": "tubetranscript"
                            },
                            signal: summaryAbortController.signal
                        });

                        const data = await response.json();

                        // API error
                        if (data.code) {
                            throw { code: data.code, message: data.message };
                        }

                        // Check summary status
                        if (data.summaries && data.summaries[0]) {
                            const summaryData = data.summaries[0];

                            // Summary is ready
                            if (summaryData.status === "READY" && summaryData.data) {
                                renderSummary(summaryData.data, videoId);
                                gtag('event', 'summary_success_mobile');
                                return;
                            }

                            // Update progress if available
                            if (summaryData.progressPercent !== undefined) {
                                updateSummaryProgress(summaryData.progressPercent);
                            }
                        }

                        // Wait before next poll
                        await new Promise(resolve => setTimeout(resolve, pollInterval));
                    }

                    // Timeout after max attempts
                    throw { code: "TIMEOUT" };
                } catch (error) {
                    if (error.name === 'AbortError') {
                        // Request was aborted, do nothing
                        return;
                    }
                    handleSummaryError(error);
                }
            }

            // Simple Markdown parser for summary content
            function parseMarkdown(text) {
                if (!text) return '';

                // Escape HTML first to prevent XSS
                let html = text
                    .replace(/&/g, '&amp;')
                    .replace(/</g, '&lt;')
                    .replace(/>/g, '&gt;');

                // Bold: **text** or __text__
                html = html.replace(/\*\*(.+?)\*\*/g, '<strong>$1</strong>');
                html = html.replace(/__(.+?)__/g, '<strong>$1</strong>');

                // Italic: *text* or _text_ (but not inside words)
                html = html.replace(/(?<!\w)\*([^*]+?)\*(?!\w)/g, '<em>$1</em>');
                html = html.replace(/(?<!\w)_([^_]+?)_(?!\w)/g, '<em>$1</em>');

                // Inline code: `code`
                html = html.replace(/`([^`]+?)`/g, '<code>$1</code>');

                // Line breaks
                html = html.replace(/\n/g, '<br>');

                return html;
            }

            // Truncate text to first N words with ellipsis
            function truncateWords(text, wordCount, addEllipsis = true) {
                const words = text.split(/\s+/);
                if (words.length <= wordCount) return text;
                return words.slice(0, wordCount).join(' ') + (addEllipsis ? '...' : '');
            }

            function truncateWithFade(text, wordCount, fadeWords = 3) {
                const words = text.split(/\s+/);
                if (words.length <= wordCount) return text;
                const visibleWords = words.slice(0, wordCount);
                const normalWords = visibleWords.slice(0, -fadeWords).join(' ');
                const fadingWords = visibleWords.slice(-fadeWords).join(' ');
                return normalWords + ' <span class="fade-text">' + fadingWords + '</span>';
            }

            let savedProTranscription = null;

            // Error messages mapping for Pro Transcription
            const proTranscriptionErrorMessages = {
                'P_QUEUE_FULL': "Aguarde a conclusão da transcrição anterior",
                'PG_QUEUE_FULL': "O servidor está ocupado, tente novamente mais tarde",
                'LIMIT_REACHED': "Limite diário atingido. Tente novamente amanhã",
                'TIMEOUT': "A transcrição está demorando muito",
                'FAILED_DOWNLOADING': "A transcrição falhou. Tente outro vídeo",
                'FAILED_TRANSCRIBING': "A transcrição falhou. Tente outro vídeo",
                'AUDIO_FORMATS': "A transcrição falhou. Tente outro vídeo"
            };

            // Errors without retry button
            const proTranscriptionNoRetryErrors = ['LIMIT_REACHED'];

            // Pro Transcription progress elements
            const proTranscriptionPercentElement = document.getElementById('pro-transcription-percent');
            const proTranscriptionProgressCircle = document.getElementById('pro-transcription-progress');

            function updateProTranscriptionProgress(percent) {
                percent = Math.min(100, Math.max(0, percent));
                if (proTranscriptionPercentElement) {
                    proTranscriptionPercentElement.textContent = Math.round(percent) + '%';
                }
                if (proTranscriptionProgressCircle) {
                    const offset = circumference - (percent / 100) * circumference;
                    proTranscriptionProgressCircle.style.strokeDashoffset = offset;
                }
            }

            // Start Pro Transcription (called from button click)
            function startProTranscription() {
                if (proTranscriptionStarted) return;
                proTranscriptionStarted = true;
                const videoId = "Ir8dALr-e-4";
                fetchProTranscription(videoId);
                gtag('event', 'pro_transcription_start');
            }

            // Main function to fetch Pro Transcription via Partner API
            async function fetchProTranscription(videoId) {
                const maxAttempts = 120; // 10 minutes max (120 * 5s = 600s)
                const pollInterval = 5000; // 5 seconds
                const apiURL = "https://yt-to-text.com/api";

                // Show loading, hide everything else
                $('#pro-transcription-promo').hide();
                $('#pro-transcription-error').hide();
                $('#pro-transcription-result').hide();
                $('#pro-transcription-loading').show();
                $('#pro-transcript-error-promo').hide();


                // Reset progress
                updateProTranscriptionProgress(0);

                // Create abort controller for cleanup
                proTranscriptionAbortController = new AbortController();

                try {
                    for (let attempt = 0; attempt < maxAttempts; attempt++) {
                        const response = await fetch(apiURL + "/p/v1/GetTranscripts", {
                            method: "POST",
                            body: JSON.stringify({ video_id: videoId }),
                            headers: {
                                "Content-Type": "application/json",
                                "X-App-Version": "tubetranscript"
                            },
                            signal: proTranscriptionAbortController.signal
                        });

                        const data = await response.json();

                        // API error
                        if (data.code) {
                            throw { code: data.code, message: data.message };
                        }

                        // Check transcription status
                        if (data.transcriptions && data.transcriptions[0]) {
                            const transcriptionData = data.transcriptions[0];

                            // Check for failed status
                            if (transcriptionData.status && transcriptionData.status.startsWith('FAILED')) {
                                throw { code: transcriptionData.status };
                            }

                            // Transcription is ready
                            if (transcriptionData.status === "READY" && transcriptionData.data) {
                                renderProTranscription(transcriptionData.data, videoId);
                                gtag('event', 'pro_transcription_success');
                                return;
                            }

                            // Update progress and status text based on status
                            const statusTextElement = document.getElementById('pro-transcription-loading-status');
                            if (transcriptionData.status === "DOWNLOADING" || transcriptionData.status === "CONVERTING") {
                                // During download/convert, use preparePercent (0-50% of total progress)
                                if (statusTextElement) {
                                    statusTextElement.textContent = "Preparando transcrição";
                                }
                                if (transcriptionData.preparePercent !== undefined) {
                                    updateProTranscriptionProgress(transcriptionData.preparePercent * 0.5);
                                }
                            } else if (transcriptionData.status === "TRANSCRIBING") {
                                // During transcription, use progressPercent (50-100% of total progress)
                                if (statusTextElement) {
                                    statusTextElement.textContent = "Gerando transcrição";
                                }
                                if (transcriptionData.progressPercent !== undefined) {
                                    updateProTranscriptionProgress(50 + transcriptionData.progressPercent * 0.5);
                                }
                            }
                        }

                        // Wait before next poll
                        await new Promise(resolve => setTimeout(resolve, pollInterval));
                    }

                    // Timeout after max attempts
                    throw { code: "TIMEOUT" };
                } catch (error) {
                    if (error.name === 'AbortError') {
                        // Request was aborted, do nothing
                        return;
                    }
                    handleProTranscriptionError(error);
                }
            }

            // Render Pro Transcription result
            function renderProTranscription(data, videoId) {
                $('#pro-transcription-loading').hide();

                if (data.transcripts && data.transcripts.length > 0) {
                    savedProTranscription = data.transcripts;
                    const transcript = getHTMLFromRawTranscript(data.transcripts, videoId);
                    $('#pro-transcription-text').html(transcript);
                    $('#pro-transcription-result').show();
                    summaryLoaded = false;
                } else {
                    // No transcripts in data
                    handleProTranscriptionError({ code: 'FAILED_TRANSCRIBING' });
                }
            }

            // Handle Pro Transcription errors
            function handleProTranscriptionError(error) {
                $('#pro-transcription-loading').hide();

                // Show error message
                const message = proTranscriptionErrorMessages[error.code] || "Falha ao gerar a transcrição";
                $('#pro-transcription-error-message').text(message);

                // Show/hide Retry button
                if (proTranscriptionNoRetryErrors.includes(error.code)) {
                    $('#pro-transcription-retry-btn').hide();
                } else {
                    $('#pro-transcription-retry-btn').show();
                }

                $('#pro-transcription-error').show();
                $('#pro-transcript-error-promo').show();
                gtag('event', 'pro_transcription_error', { error_code: error.code || 'unknown' });
            }

            // Retry Pro Transcription
            function retryProTranscription() {
                const videoId = "Ir8dALr-e-4";
                proTranscriptionStarted = false;
                startProTranscription();
            }

            // Copy Pro Transcription text
            function copyProTranscript(withTimecodes) {
                const proTranscriptContent = document.getElementById("pro-transcription-text");
                if (proTranscriptContent === null) {

                let contentBody = "";

                const url = `https://www.youtube.com/watch?v=${videoId}`;

                Swal.fire({

                    toast: true
                });
            }

            // Download Pro Transcription subtitles
            function downloadProSubtitles(format) {
                if (!savedProTranscription) {
                    return;
                }

                function formatTimeForDownload(seconds, format) {
                    const hh = String(Math.floor(seconds / 3600)).padStart(2, '0');
                    const mm = String(Math.floor((seconds % 3600) / 60)).padStart(2, '0');
                    const ss = String(Math.floor(seconds % 60)).padStart(2, '0');
                    const ms = String(Math.round((seconds % 1) * 1000)).padStart(3, '0');
                    const separator = format === 'srt' ? ',' : '.';
                    return `${hh}:${mm}:${ss}${separator}${ms}`;
                }

                let content = '';
                const ext = format.toLowerCase();

                if (ext === 'vtt') {
                    content = 'WEBVTT\n\n';
                    savedProTranscription.forEach((sub) => {
                        const start = formatTimeForDownload(parseFloat(sub.s), 'vtt');
                        const end = formatTimeForDownload(parseFloat(sub.e), 'vtt');
                        content += `${start} --> ${end}\n${sub.t}\n\n`;
                    });
                } else if (ext === 'srt') {
                    savedProTranscription.forEach((sub, index) => {
                        const start = formatTimeForDownload(parseFloat(sub.s), 'srt');
                        const end = formatTimeForDownload(parseFloat(sub.e), 'srt');
                        content += `${index + 1}\n${start} --> ${end}\n${sub.t}\n\n`;
                    });
                } else if (ext === 'txt') {
                    const videoId = "Ir8dALr-e-4";
                    const url = `https://www.youtube.com/watch?v=${videoId}`;
                    content += `${url}\n\nTranscript:\n`;
                    savedProTranscription.forEach((sub) => {
                        content += `${sub.t}\n`;
                    });
                } else {
                    return;
                }

                const blob = new Blob([content], { type: 'text/plain;charset=utf-8' });
                const url = URL.createObjectURL(blob);
                const link = document.createElement('a');
                link.href = url;
                link.download = `Ir8dALr-e-4_pro.${ext}`;
                document.body.appendChild(link);
                link.click();
                document.body.removeChild(link);
                URL.revokeObjectURL(url);
            }

            // Switch to Pro Transcription tab and start generation
            function switchToProTranscriptionAndStart() {
                $('#pro-transcription-tab').tab('show');
                // Small delay to ensure tab is shown before starting
                setTimeout(() => {
                    startProTranscription();
                }, 100);
            }
