            // localStorage helpers for transcript view counting
            function getViewsToday(key, today) {
                try {
                    const raw = localStorage.getItem(key);
                    if (!raw) return { date: '', count: 0, videoIds: [] };
                    const data = JSON.parse(raw);
                    if (data.date !== today) return { date: '', count: 0, videoIds: [] };
                    return data;
                } catch (e) {
                    return { date: '', count: 0, videoIds: [] };
                }
            }

            function recordView(key, videoId, today) {
                try {
                    const views = getViewsToday(key, today);
                    if (views.date === today && views.videoIds.includes(videoId)) return views.count;
                    const newData = {
                        date: today,
                        count: (views.date === today ? views.count : 0) + 1,
                        videoIds: views.date === today ? [...views.videoIds, videoId] : [videoId]
                    };
                    localStorage.setItem(key, JSON.stringify(newData));
                    return newData.count;
                } catch (e) {
                    return 0;
                }
            }

            function shouldApplyLimit(key, videoId) {
                const today = new Date().toISOString().slice(0, 10);
                const views = getViewsToday(key, today);

                // If user has already viewed transcript for this video today, don't apply limit
                if (views.date === today && views.videoIds.includes(videoId)) return false;

                if (views.count >= GEO_LIMIT.maxFreeViews) {
                    // User has already exceeded the limit today
                    return true;
                }

                const count = recordView(key, videoId, today);
                return false;
            }

            function downloadSubtitles(format) {
                if (!savedTranscription) {
                    return;
                }
                function formatTime(seconds, format) {
                    const date = new Date(seconds * 1000);
                    const hh = String(Math.floor(seconds / 3600)).padStart(2, '0');
                    const mm = String(Math.floor((seconds % 3600) / 60)).padStart(2, '0');
                    const ss = String(Math.floor(seconds % 60)).padStart(2, '0');
                    const ms = String(Math.round((seconds % 1) * 1000)).padStart(3, '0');

                    const separator = format === 'srt' ? ',' : '.';
                    return `${hh}:${mm}:${ss}${separator}${ms}`;
                }

                let content = '';
                const ext = format.toLowerCase();

                if (ext === 'vtt') {
                    content = 'WEBVTT\n\n';

                    savedTranscription.forEach((sub, index) => {
                        const start = formatTime(parseFloat(sub.s), 'vtt');
                        const end = formatTime(parseFloat(sub.e), 'vtt');
                        content += `${start} --> ${end}\n${sub.t}\n\n`;
                    });
                } else if (ext === 'srt') {
                    savedTranscription.forEach((sub, index) => {
                        const start = formatTime(parseFloat(sub.s), 'srt');
                        const end = formatTime(parseFloat(sub.e), 'srt');
                        content += `${index + 1}\n${start} --> ${end}\n${sub.t}\n\n`;
                    });
                } else if (ext === 'txt') {
                    content = extractTranscriptFromHTML(true)
                } else {
                    throw new Error('Unsupported format. Use "srt" or "vtt" or "txt"');
                }

                const blob = new Blob([content], { type: 'text/plain;charset=utf-8' });
                const url = URL.createObjectURL(blob);

                const link = document.createElement('a');
                link.href = url;
                link.download = `cakomLEi00c.${ext}`;
                document.body.appendChild(link);
                link.click();
                document.body.removeChild(link);

                URL.revokeObjectURL(url);
            }

                return minutes + ":" + (remainingSeconds < 10 ? '0' : '') + remainingSeconds;
            }

            function getHTMLFromRawTranscript(rawTranscript, videoId) {

                const scriptObjArr = [], timeUpperLimit = 60, charInitLimit = 300, charUpperLimit = 500;
                let loop = 0, chars = [], charCount = 0, timeSum = 0, tempObj = {}, remaining = {};

                // Sum-up to either total 60 seconds or 300 chars.
                Array.from(rawTranscript).forEach((obj, i, arr) => {
                    if (obj.s !== undefined && obj.t !== undefined) {
                        obj.start = obj.s;
                        obj.text = obj.t;
                    }

                    // Check Remaining Text from Prev Loop
                    if (remaining.start && remaining.text) {
                        tempObj.start = remaining.start;
                        chars.push(remaining.text);
                        remaining = {}; // Once used, reset to {}
                    }

                    // Initial Loop: Set Start Time
                    if (loop == 0) {
                        tempObj.start = (remaining.start) ? remaining.start : obj.start;
                    }

                    loop++;

                    const startSeconds = Math.round(tempObj.start);
                    const seconds = Math.round(obj.start);
                    timeSum = (seconds - startSeconds);
                    charCount += obj.text.length;
                    chars.push(obj.text);

                    if (i == arr.length - 1) {
                        tempObj.text = chars.join(" ").replace(/\n/g, " ");
                        scriptObjArr.push(tempObj);
                        resetNums();
                        return;
                    }

                    if (timeSum > timeUpperLimit) {
                        tempObj.text = chars.join(" ").replace(/\n/g, " ");
                        scriptObjArr.push(tempObj);
                        resetNums();
                        return;
                    }

                    if (charCount > charInitLimit) {

                        if (charCount < charUpperLimit) {
                            if (obj.text.includes(".")) {

                                const splitStr = obj.text.split(".");

                                // Case: the last letter is . => Process regulary
                                if (splitStr[splitStr.length - 1].replace(/\s+/g, "") == "") {
                                    tempObj.text = chars.join(" ").replace(/\n/g, " ");
                                    scriptObjArr.push(tempObj);
                                    resetNums();
                                    return;
                                }

                                // Case: . is in the middle
                                // 1. Get the (length - 2) str, then get indexOf + str.length + 1, then substring(0,x)
                                // 2. Create remaining { text: str.substring(x), start: obj.start } => use the next loop
                                const lastText = splitStr[splitStr.length - 2];
                                const substrIndex = obj.text.indexOf(lastText) + lastText.length + 1;
                                const textToUse = obj.text.substring(0, substrIndex);
                                remaining.text = obj.text.substring(substrIndex);
                                remaining.start = obj.start;

                                // Replcae arr element
                                chars.splice(chars.length - 1, 1, textToUse)
                                tempObj.text = chars.join(" ").replace(/\n/g, " ");
                                scriptObjArr.push(tempObj);
                                resetNums();
                                return;

                            } else {
                                // Move onto next loop to find .
                                return;
                            }
                        }

                        tempObj.text = chars.join(" ").replace(/\n/g, " ");
                        scriptObjArr.push(tempObj);
                        resetNums();
                    }
                })

                return Array.from(scriptObjArr).map((obj, idx) => {
                    const t = Math.round(obj.start);
                    const hhmmss = convertIntToHms(t);
                    const timeDisplay = videoId
                        ? `<a href="https://youtube.com/watch?v=${videoId}&t=${t}s" target="_blank">${hhmmss}</a>`
                        : hhmmss;
                    const html = `
                    <div class="transcript-group-box">
                        <div class="transcript-time">${timeDisplay}</div>
                        <div class="transcript-text">${obj.text}</div>
                    </div>`
                    return html
                }).join("");

                function resetNums() {
                    loop = 0, chars = [], charCount = 0, timeSum = 0, tempObj = {};
                }

                function convertIntToHms(num) {
                    const h = (num < 3600) ? 14 : 12;
                    return (new Date(num * 1000).toISOString().substring(h, 19)).toString();
                }
            }

            let currentPercent = 0;
            let animationInterval;
            let pulsingStarted = false;

            const percentElement = document.getElementById('percent');
            const progressCircle = document.getElementById('progress');
            const loadingStatus = document.getElementById('loading-status');
            const dotsElement = document.getElementById('dots');

            const radius = 90;
            const circumference = 2 * Math.PI * radius;

            function updateProgress(percent) {
                percent = Math.min(100, Math.max(0, percent));

                percentElement.textContent = Math.round(percent) + '%';

                const offset = circumference - (percent / 100) * circumference;
                progressCircle.style.strokeDashoffset = offset;

                if (percent >= 95 && !pulsingStarted) {
                    pulsingStarted = true;
                    startPulsing();
                }
            }

            function startPulsing() {
                // Добавляем классы для мерцания
                progressCircle.classList.add('pulsing');
                percentElement.classList.add('pulsing');
                dotsElement.classList.add('animating');

                loadingStatus.textContent = "Quase pronto";
            }

            function stopPulsing() {
                progressCircle.classList.remove('pulsing');
                percentElement.classList.remove('pulsing');
                dotsElement.classList.remove('animating');
                pulsingStarted = false;
            }

            function startAnimation() {
                currentPercent = 0;
                updateProgress(0);
                loadingStatus.textContent = "Carregando";
                stopPulsing();

                if (animationInterval) {
                    clearInterval(animationInterval);
                }

                const targetPercent = 95;
                const duration = 10000;
                const interval = 50;
                const steps = duration / interval;
                const increment = targetPercent / steps;

                animationInterval = setInterval(() => {
                    currentPercent += increment;

                    if (currentPercent >= targetPercent) {
                        currentPercent = targetPercent;
                        updateProgress(currentPercent);
                        clearInterval(animationInterval);
                    } else {
                        updateProgress(currentPercent);
                    }
                }, interval);
            }

            // Simple Markdown parser for summary content
            function parseMarkdown(text) {
                if (!text) return '';

                // Escape HTML first to prevent XSS
                let html = text
                    .replace(/&/g, '&amp;')
                    .replace(/</g, '&lt;')
                    .replace(/>/g, '&gt;');

                // Bold: **text** or __text__
                html = html.replace(/\*\*(.+?)\*\*/g, '<strong>$1</strong>');
                html = html.replace(/__(.+?)__/g, '<strong>$1</strong>');

                // Italic: *text* or _text_ (but not inside words)
                html = html.replace(/(?<!\w)\*([^*]+?)\*(?!\w)/g, '<em>$1</em>');
                html = html.replace(/(?<!\w)_([^_]+?)_(?!\w)/g, '<em>$1</em>');

                // Inline code: `code`
                html = html.replace(/`([^`]+?)`/g, '<code>$1</code>');

                // Line breaks
                html = html.replace(/\n/g, '<br>');

                return html;
            }

            // Truncate text to first N words with ellipsis
            function truncateWords(text, wordCount, addEllipsis = true) {
                const words = text.split(/\s+/);
                if (words.length <= wordCount) return text;
                return words.slice(0, wordCount).join(' ') + (addEllipsis ? '...' : '');
            }

            function truncateWithFade(text, wordCount, fadeWords = 3) {
                const words = text.split(/\s+/);
                if (words.length <= wordCount) return text;
                const visibleWords = words.slice(0, wordCount);
                const normalWords = visibleWords.slice(0, -fadeWords).join(' ');
                const fadingWords = visibleWords.slice(-fadeWords).join(' ');
                return normalWords + ' <span class="fade-text">' + fadingWords + '</span>';
            }

            // Download Pro Transcription subtitles
            function downloadProSubtitles(format) {
                if (!savedProTranscription) {
                    return;
                }

                function formatTimeForDownload(seconds, format) {
                    const hh = String(Math.floor(seconds / 3600)).padStart(2, '0');
                    const mm = String(Math.floor((seconds % 3600) / 60)).padStart(2, '0');
                    const ss = String(Math.floor(seconds % 60)).padStart(2, '0');
                    const ms = String(Math.round((seconds % 1) * 1000)).padStart(3, '0');
                    const separator = format === 'srt' ? ',' : '.';
                    return `${hh}:${mm}:${ss}${separator}${ms}`;
                }

                let content = '';
                const ext = format.toLowerCase();

                if (ext === 'vtt') {
                    content = 'WEBVTT\n\n';
                    savedProTranscription.forEach((sub) => {
                        const start = formatTimeForDownload(parseFloat(sub.s), 'vtt');
                        const end = formatTimeForDownload(parseFloat(sub.e), 'vtt');
                        content += `${start} --> ${end}\n${sub.t}\n\n`;
                    });
                } else if (ext === 'srt') {
                    savedProTranscription.forEach((sub, index) => {
                        const start = formatTimeForDownload(parseFloat(sub.s), 'srt');
                        const end = formatTimeForDownload(parseFloat(sub.e), 'srt');
                        content += `${index + 1}\n${start} --> ${end}\n${sub.t}\n\n`;
                    });
                } else if (ext === 'txt') {
                    const videoId = "cakomLEi00c";
                    const url = `https://www.youtube.com/watch?v=${videoId}`;
                    content += `${url}\n\nTranscript:\n`;
                    savedProTranscription.forEach((sub) => {
                        content += `${sub.t}\n`;
                    });
                } else {
                    return;
                }

                const blob = new Blob([content], { type: 'text/plain;charset=utf-8' });
                const url = URL.createObjectURL(blob);
                const link = document.createElement('a');
                link.href = url;
                link.download = `cakomLEi00c_pro.${ext}`;
                document.body.appendChild(link);
                link.click();
                document.body.removeChild(link);
                URL.revokeObjectURL(url);
            }
